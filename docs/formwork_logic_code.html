🔢 Bước 2: Hàm Tính Chính (Code Thực Tế, Nguyên Văn)

public static void CalculateFormWorkArea_Total(this ColumnR column) { // If element has solid then start calculat formwork if (column.GeometryData.InstanceSolid != null) { double fw_Side = 0; double fw_Sub_Beam = 0; double fw_Sub_Found = 0; double fw_Sub_Slab = 0; double fw_Sub_Col = 0; double fw_Sub_Wall = 0; double fw_Sub_Gen = 0; List<Face> computedFaces = column.GeometryData.Faces; List<Element> intersectingElements = column.IntersectingElements; List<Face> computedSideFaces = new List<Face>(); if (column.IsSlanted) { XYZ colDir = (column.LocationCurve as Line).Direction; computedSideFaces = computedFaces.Where(x => x.ComputeNormal(new UV(0.5, 0.5)).DotProduct(colDir).IsEqual(0)).ToList(); } else { computedSideFaces = computedFaces.Where(x => x.ComputeNormal(new UV(0.5, 0.5)).DotProduct(XYZ.BasisZ).IsEqual(0)).ToList(); } foreach (Face elemFace in computedSideFaces) { fw_Side += elemFace.Area; foreach (Element elem in intersectingElements) { // Retrieve GeometryData from element collection GeometryData elemGeom = elementCollection[elem.Id.AsValue()].GeometryData; // Element surely has instance solid because of checking intersecting elements step if (elemGeom.InstanceSolid != null) { foreach (Face checkface in elemGeom.Faces) { Tuple<bool, double, bool> overlappingResult = elemFace.OverlappingResult(checkface); if (overlappingResult.Item3) // failed calculating overlap { warningIDPairCollection.AddIdsToList(column.RevitElement, elem); } else if (overlappingResult.Item1) { AssignOverlapAreaIntoCategory(elem, overlappingResult.Item2, ref fw_Sub_Found, ref fw_Sub_Beam, ref fw_Sub_Slab, ref fw_Sub_Col, ref fw_Sub_Wall, ref fw_Sub_Gen); fw_Side -= overlappingResult.Item2; } } } } } SetFormworkDataSide(column.FormworkData, fw_Side, fw_Sub_Found, fw_Sub_Beam, fw_Sub_Slab, fw_Sub_Col, fw_Sub_Wall, fw_Sub_Gen); } }

Hàm phụ: Phân Loại Overlap Theo Category

File: OOG.STRPlugin/Utilities/FormworkUtils.cs (dòng 96-127)

private static void AssignOverlapAreaIntoCategory(Element e, double value, ref double found, ref double beam, ref double slab, ref double col, ref double wall, ref double gen) { Category cat = e.Category; if (cat != null) { BuiltInCategory bic = (BuiltInCategory)cat.Id.AsValue(); if (bic == BuiltInCategory.OST_StructuralFoundation) { found += value; } else if (bic == BuiltInCategory.OST_StructuralFraming) { beam += value; } else if (bic == BuiltInCategory.OST_Floors) { slab += value; } else if (bic == BuiltInCategory.OST_StructuralColumns) { col += value; } else if (bic == BuiltInCategory.OST_Walls) { wall += value; } else if (bic == BuiltInCategory.OST_GenericModel) { gen += value; } } }

private static void SetFormworkDataSide(FormworkData fwData, double side, double found, double beam, double slab, double col, double wall, double gen) { fwData.FW_Side += side.Round(3); fwData.FW_Side = fwData.FW_Side < 0 ? 0 : fwData.FW_Side; fwData.FW_Sub_Found += found.Round(3); fwData.FW_Sub_Found = fwData.FW_Sub_Found < 0 ? 0 : fwData.FW_Sub_Found; fwData.FW_Sub_Beam += beam.Round(3); fwData.FW_Sub_Beam = fwData.FW_Sub_Beam < 0 ? 0 : fwData.FW_Sub_Beam; // ... tương tự cho Slab, Col, Wall, Gen }

🔢 Bước 3: Hàm Tính Overlap (Code Thực Tế, Nguyên Văn)

File: Core.Revit/BiMoLib/Revit/Utils/FaceUtil.cs (dòng 123-323, phần chính cho Planar Face)

public static Tuple<bool, double, bool> OverlappingResult(this Face _f1, Face _f2) { bool check = false; double area = 0; bool failed = false; double extrudeDist = 0.1; double offsetDist1 = 0.005; double offsetDist2 = 0.001; if (_f1.IsPlanarFaceOnSamePlane(_f2)) { IList<CurveLoop> _f1CurveLoops = _f1.GetEdgesAsCurveLoops(); IList<CurveLoop> _f2CurveLoops = _f2.GetEdgesAsCurveLoops(); XYZ extrudeDir = _f1.ComputeNormal(new UV(0.5, 0.5)); try // Execute normal checking process { Solid solid1 = _f1CurveLoops.Try_ExtrudeCurveLoops(extrudeDir, extrudeDist); Solid solid2 = _f2CurveLoops.Try_ExtrudeCurveLoops(extrudeDir, extrudeDist); Solid intersectResult = BooleanOperationsUtils.ExecuteBooleanOperation(solid1, solid2, BooleanOperationsType.Intersect); double volume = Math.Abs(intersectResult.Volume); if (volume > 0.000001) { check = true; area = volume / extrudeDist; } } catch // If the process has error, try to create slightly offset curveloops and run checking. If still has error, return failed { try { double volume = 0; IList<CurveLoop> _f1CurveLoops_Offset = new List<CurveLoop>(); foreach (CurveLoop cl in _f1CurveLoops) { _f1CurveLoops_Offset.Add(CurveLoop.CreateViaOffset(cl, offsetDist1, _f1.ComputeNormal(new UV(0.5, 0.5)))); } IList<CurveLoop> _f2CurveLoops_Offset = new List<CurveLoop>(); foreach (CurveLoop cl in _f2CurveLoops) { _f2CurveLoops_Offset.Add(CurveLoop.CreateViaOffset(cl, offsetDist2, _f2.ComputeNormal(new UV(0.5, 0.5)))); } Solid solid1 = _f1CurveLoops_Offset.Try_ExtrudeCurveLoops(extrudeDir, extrudeDist); Solid solid2 = _f2CurveLoops_Offset.Try_ExtrudeCurveLoops(extrudeDir, extrudeDist); Solid intersectResult = BooleanOperationsUtils.ExecuteBooleanOperation(solid1, solid2, BooleanOperationsType.Intersect); volume = Math.Abs(intersectResult.Volume); if (volume > 0.000001) { check = true; area = volume / extrudeDist; } } catch { failed = true; check = false; } } } // ... (nhánh else if xử lý Cylindrical Face bị lược bớt cho gọn) else { // 1 planar and 1 curve face will return false } return new Tuple<bool, double, bool>(check, area, failed); }

Hàm phụ điều kiện: Kiểm Tra 2 Mặt Cùng Mặt Phẳng

File: Core.Revit/BiMoLib/Revit/Utils/FaceUtil.cs (dòng 16-35)

public static bool IsPlanarFaceOnSamePlane(this Face _f1, Face _f2) { if (_f1 is PlanarFace && _f2 is PlanarFace) { XYZ f1_normal = _f1.ComputeNormal(new UV(0.5, 0.5)); XYZ f2_normal = _f2.ComputeNormal(new UV(0.5, 0.5)); XYZ pt0 = XYZ.Zero; XYZ pt1 = _f1.Evaluate(new UV(0.5, 0.5)); XYZ pt2 = _f2.Evaluate(new UV(0.5, 0.5)); double dist1 = Math.Round(Math.Abs(pt0.Subtract(pt1).DotProduct(f1_normal)), 3); double dist2 = Math.Round(Math.Abs(pt0.Subtract(pt2).DotProduct(f2_normal)), 3); bool check = (f1_normal.IsParallel(f2_normal) && dist1 == dist2) ? true : false; return check; } else { return false; } }

→ 2 mặt được coi là "cùng mặt phẳng" khi normal của chúng song song (IsParallel) VÀ khoảng cách từ gốc tọa độ tới mỗi mặt (chiếu theo normal) bằng nhau.

📊 Ví Dụ Thực Tế: Cột + Dầm

Input Data

Column: - 4 vertical faces: mỗi 0.4m × 4m = 1.6 m² - Tổng = 6.4 m² - IntersectingElements = [Beam_1] Beam: - Xuyên qua cột ở độ cao 1.5m - Size: 0.3m × 0.5m - Overlap = 0.15 m²

Execution Trace

CalculateFormWorkArea_Total(column): ├─ fw_Side = 0 ├─ computedSideFaces = [Face_1, Face_2, Face_3, Face_4] │ ├─ Loop Face 1 (1.6 m²): │ ├─ fw_Side = 0 + 1.6 = 1.6 │ ├─ Check Beam_1: │ │ ├─ OverlappingResult(Face_1, Beam.Face): │ │ │ ├─ Same plane? YES │ │ │ ├─ Volume = 0.015 cubic feet │ │ │ └─ area = 0.015 / 0.1 = 0.15 m² │ │ │ │ │ ├─ result.Item1 = true (overlap) │ │ ├─ fw_Sub_Beam += 0.15 │ │ └─ fw_Side -= 0.15 → fw_Side = 1.45 │ ├─ Loop Face 2: fw_Side = 1.45 + 1.6 = 3.05 ├─ Loop Face 3: fw_Side = 3.05 + 1.6 = 4.65 ├─ Loop Face 4: fw_Side = 4.65 + 1.6 = 6.25 │ └─ SetFormworkDataSide(column.FormworkData, 6.25, ...) └─ Result: FW_Total = 6.25 m² ✓

Output

FW_Total = 6.25 m² ├─ Tổng diện tích 4 mặt: 6.4 m² ├─ Trừ overlap beam: -0.15 m² └─ Kết quả cuối: 6.25 m² ✓

🎯 Key Points (Các Điểm Chính)

  • Loop Elements: Duyệt tất cả cột, dầm, tường, ...
  • Lấy Faces: Từ 3D solid lấy ra tất cả mặt 2D
  • Lọc Vertical: Chỉ lấy mặt bên (không lấy đáy)
  • Cộng Area: Cộng dần diện tích từng mặt
  • Check Overlap: So sánh với elements giao qua
  • Boolean Intersection: Tìm phần giao → Volume → Area
  • Trừ Overlap: fw_Side -= overlap area
  • Lưu Kết Quả: SetFormworkDataSide() → Revit Parameter
